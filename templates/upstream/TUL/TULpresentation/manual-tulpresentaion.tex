\section{Práce s šablonou}
\subsection{Úvod}
\begin{frame}[fragile]
  \frametitle{Práce s šablonou}
  \framesubtitle{Úvod}
  \begin{itemize}
    \item Třída \verb|tulpresentation.cls| je založena na třídě beamer a upravena v souladu s novým vizuálním stylem Technické univerzity v Liberci (od r. 2022)
    \item Nastavení používá zavedené definice z třídy \verb|tulthesis.cls| P. Satrapy
    \item Šablona je nasdílena v online prostředí \emph{overleaf}, odtud je možné ji stáhnout a spustit offline 
    \item Odkaz na šablony je: \href{www.fm.tul.cz/personal/jan.koprnicky}{www.fm.tul.cz/personal/jan.koprnicky}
    \item Šablona vytváří logotypy součástí TUL z fontu \emph{TUL Mono Bold}, který je v projektu nebo může být nainstalován v operačním systému
    \item Doplňkový symbol TUL v bílé barvě je nezbytný pro vytvoření titulní a závěrečné stránky prezentace 
    \item Z těchto důvodů nejsou součástí šablony žádná předpřipravená loga
  \end{itemize}
\end{frame}

\subsection{Parametry třídy}
\begin{frame}[fragile]{Parametry třídy \emph{tulpresentation.cls}}
\verb|\documentclass[<parametry>]{tulpresentation}| 

  \begin{columns}[T]
      \begin{column}{0.4\textwidth}
        \begin{itemize}
          \item Typ součásti TUL
          \begin{itemize}
            % \item \verb|TUL| (výchozí)
            \item \verb|FS|
            \item \verb|FT|
            \item \verb|FP|
            \item \verb|EF|
            \item \verb|FUA|
            \item \verb|FM|
            \item \verb|FZS|
            \item \verb|CXI|
            \item Není-li parametr uveden, je nastaven \verb|TUL| (výchozí)
          \end{itemize}      
        \end{itemize}        
      \end{column}
      \begin{column}{0.6\textwidth}
        \begin{itemize}
          \item Jazyková mutace logotypů
          \begin{itemize}
            \item český jazyk (výchozí)
            \item \verb|EN| (anglický jazyk)
          \end{itemize}
          \item Barva prezentace        
          \begin{itemize}
            \item V souladu s vybranou součástí TUL (výchozí)
            \item \verb|bw| (ovlivní barvu titulní stránky, záhlaví a zápatí, nikoli barvu obsahu slajdů)
          \end{itemize}
        \end{itemize}
      \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Příklady nastavení parametrů třídy}
  \begin{itemize}
    \item Prezentace v barvách Technické univerzity v Liberci s českými logy, titulní a závěrečnou stranou:

    \verb|\documentclass[]{tulpresentation}|

    \item Prezentace v barvách Fakulty přírodně-humanitní a pedagogické s anglickými logy:
    
    \verb|\documentclass[FP,EN]{tulpresentation}|
    
    \item Prezentace černobílá Fakulty mechatroniky, informatiky a mezioborových studií s anglickými logy:

    \verb|\documentclass[FM,bw,EN]{tulpresentation}|
  \end{itemize}
\end{frame}
